    <script>
        let isEditMode = false;

        // 편집 모드 전환
        function toggleEditMode() {
            isEditMode = !isEditMode;
            alert(isEditMode ? '✏️ 편집 모드 활성화!\n항목을 클릭하여 수정하세요.' : '보기 모드로 전환되었습니다.');
        }

        // 히어로 섹션 수정
        function editHero() {
            if (!isEditMode) return;
            const title = prompt('제목 입력:', document.getElementById('heroTitle').textContent);
            if (title) document.getElementById('heroTitle').textContent = title;
            
            const desc = prompt('설명 입력:', document.getElementById('heroDesc').textContent);
            if (desc) document.getElementById('heroDesc').textContent = desc;
        }

        // 서비스 제목 수정
        function editServiceTitle() {
            if (!isEditMode) return;
            const title = prompt('서비스 섹션 제목:', '주요 서비스');
            if (title) document.querySelectorAll('.services .section-title')[0].textContent = title;
        }

        // 서비스 카드 수정
        function editService(element) {
            if (!isEditMode) return;
            const h3 = element.querySelector('h3').textContent;
            const p = element.querySelector('p').textContent;
            
            const newH3 = prompt('제목 입력:', h3);
            if (newH3) element.querySelector('h3').textContent = newH3;
            
            const newP = prompt('설명 입력:', p);
            if (newP) element.querySelector('p').textContent = newP;
        }

        // 제품 제목 수정
        function editProductTitle() {
            if (!isEditMode) return;
            const title = prompt('제품 섹션 제목:', '취급 품목');
            if (title) document.querySelectorAll('.products .section-title')[0].textContent = title;
        }

        // 제품 항목 수정
        function editProduct(element) {
            if (!isEditMode) return;
            const text = element.querySelector('p').textContent;
            const newText = prompt('제품명 입력:', text);
            if (newText) element.querySelector('p').textContent = newText;
        }

        // 연락처 수정
        function editContactTitle() {
            if (!isEditMode) return;
            const title = prompt('연락처 섹션 제목:', '📍 매장 방문 / 연락처');
            if (title) document.querySelector('.contact h2').textContent = title;
        }

        function editPhone1() {
            if (!isEditMode) return;
            const phone = prompt('매장 전화 입력:', document.getElementById('phone1').textContent);
            if (phone) {
                document.getElementById('phone1').textContent = phone;
                document.getElementById('phone1').href = 'tel:' + phone;
            }
        }

        function editPhone2() {
            if (!isEditMode) return;
            const phone = prompt('휴대폰 입력:', document.getElementById('phone2').textContent);
            if (phone) {
                document.getElementById('phone2').textContent = phone;
                document.getElementById('phone2').href = 'tel:' + phone;
            }
        }

        function editAddress() {
            if (!isEditMode) return;
            const addr = prompt('주소 입력:', document.getElementById('address').textContent.replace('<br>', ', '));
            if (addr) document.getElementById('address').innerHTML = addr.replace(', ', '<br>');
        }

        function editHours() {
            if (!isEditMode) return;
            const hours = prompt('영업시간 입력 (줄바꿈은 \\n 사용):', document.getElementById('hours').textContent.replace('<br>', '\\n'));
            if (hours) document.getElementById('hours').innerHTML = hours.replace(/\\n/g, '<br>');
        }

        function editFooter() {
            if (!isEditMode) return;
            const footer = prompt('푸터 텍스트:', document.getElementById('footerText').textContent);
            if (footer) document.getElementById('footerText').textContent = footer;
        }

        // 저장
        function saveChanges() {
            localStorage.setItem('docomData', JSON.stringify({
                heroTitle: document.getElementById('heroTitle').textContent,
                heroDesc: document.getElementById('heroDesc').textContent,
                phone1: document.getElementById('phone1').textContent,
                phone2: document.getElementById('phone2').textContent,
                address: document.getElementById('address').innerHTML,
                hours: document.getElementById('hours').innerHTML,
                footerText: document.getElementById('footerText').textContent
            }));
            alert('💾 변경사항이 저장되었습니다!');
        }

        // 초기화
        function resetChanges() {
            if (confirm('정말 초기 상태로 돌아가시겠습니까?')) {
                localStorage.removeItem('docomData');
                location.reload();
            }
        }

        // 페이지 로드 시 저장된 데이터 복원
        window.onload = function() {
            const data = localStorage.getItem('docomData');
            if (data) {
                const parsed = JSON.parse(data);
                document.getElementById('heroTitle').textContent = parsed.heroTitle;
                document.getElementById('heroDesc').textContent = parsed.heroDesc;
                document.getElementById('phone1').textContent = parsed.phone1;
                document.getElementById('phone2').textContent = parsed.phone2;
                document.getElementById('address').innerHTML = parsed.address;
                document.getElementById('hours').innerHTML = parsed.hours;
                document.getElementById('footerText').textContent = parsed.footerText;
            }
        }
    </script>
